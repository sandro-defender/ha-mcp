#include <tunables/global>

profile ha_agent_gateway flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # ---------------------------------------------------------------------------
  # Network: listeners (Ingress 8099, agent 8098) and outbound connectors.
  # ---------------------------------------------------------------------------
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,

  # ---------------------------------------------------------------------------
  # Capabilities needed before the privilege drop (run.sh).
  #
  # The container starts as root only to fetch the Supervisor-mounted /data
  # volume, then immediately execs su-exec to become the unprivileged `app`
  # user.  The Python process itself runs without capabilities.
  # ---------------------------------------------------------------------------
  capability chown,
  capability dac_override,
  capability setgid,
  capability setuid,

  # ---------------------------------------------------------------------------
  # S6-Overlay v3 startup chain (the HA base image owns ENTRYPOINT /init).
  #
  # Root cause of the 0.3.x startup failures:
  #   /bin/sh: can't open '/init': Permission denied
  #   /bin/sh: can't open '/package/admin/s6-overlay-*/libexec/preinit': Permission denied
  #   s6-overlay-suexec: fatal: child failed with exit code 2
  #
  # /init, libexec/preinit and libexec/stage0 are `#!/bin/sh -e` scripts.  The
  # kernel handles the shebang, so it starts `/bin/sh <script>` and the
  # interpreter must OPEN the file for reading.  An execute-only rule (`ix`)
  # therefore still fails with EACCES inside /bin/sh -- which is exactly the
  # error above -- because AppArmor mediates the interpreter's open(2).
  # Every S6 script needs read + inherit-execute (`rix`).
  # ---------------------------------------------------------------------------
  /init rix,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** rix,
  /etc/fix-attrs.d/** r,

  # /run.sh is the add-on's CMD.  rc.init (the stage-2 script) executes it, so
  # it needs the same read+execute pair as the S6 scripts -- the /run/{,**}
  # rule below covers the /run directory tree, not this sibling file.  Missing
  # this rule produced, in CI under the real profile:
  #   /run/s6/basedir/scripts/rc.init: line 105: /run.sh: Permission denied
  /run.sh rix,

  # Stage-2 legacy hooks (/etc/cont-init.d, /etc/services.d) and their finish
  # counterparts.  Scripts executed through a shebang need `r` as well.
  /etc/services.d/** rix,
  /etc/cont-init.d/** rix,
  /etc/cont-finish.d/** rix,

  # s6-linux-init compiles its runtime supervision tree under /run/s6 and
  # executes /run/s6/basedir/bin/init (itself a shell script).
  /run/{s6,s6-rc*,service}/** rwkix,
  /run/{,**} rwk,
  /dev/tty rw,

  # ---------------------------------------------------------------------------
  # Interpreters and binaries.
  #
  # On Alpine /bin/sh, /bin/chown, /bin/mkdir ... are symlinks to /bin/busybox
  # and AppArmor mediates the resolved path, so rules are granted per tree
  # instead of per applet.
  # ---------------------------------------------------------------------------
  /bin/** mrix,
  /sbin/** mrix,
  /usr/bin/** mrix,
  /usr/sbin/** mrix,
  /usr/local/bin/** mrix,

  # Shared libraries and CPython extensions are mmap'ed with PROT_EXEC, which
  # requires the `m` permission (`r` alone is not enough).
  /lib/** mr,
  /lib64/** mr,
  /usr/lib/** mr,
  /usr/libexec/** mr,
  /usr/local/lib/** mr,
  /opt/venv/** mr,
  /opt/venv/bin/** mrix,
  /app/** r,

  # ---------------------------------------------------------------------------
  # Application state and temporary files.  `k` (file locking) is required by
  # SQLite WAL locks on the audit/approval database.
  # ---------------------------------------------------------------------------
  /data/** rwk,
  /data/ rwk,
  /ssl/** r,
  /tmp/** rwk,
  /tmp/ rwk,

  # ---------------------------------------------------------------------------
  # Read-only system paths required by libc, CPython, OpenSSL and s6.
  # Nothing outside /data, /tmp and /run is writable.
  # ---------------------------------------------------------------------------
  /etc/** r,
  # Directory entries need their own rule: `/proc/**` does not match `/proc/`,
  # and a process table that cannot be listed breaks `ps`, `top`, and -- more
  # importantly -- s6-linux-init's shutdown path, which enumerates /proc to reap
  # remaining processes.  The container then never exits and Docker has to
  # SIGKILL it (exit 137).
  /proc/ r,
  /proc/** r,
  /sys/ r,
  /sys/devices/system/cpu/** r,
  /usr/share/** r,
  /dev/urandom r,
  /dev/random r,
  /dev/null rw,
  /dev/zero rw,
  /dev/console rw,
  /dev/pts/** rw,
  /dev/shm/ rw,
  /dev/shm/** rw,

  # ---------------------------------------------------------------------------
  # Process management from the S6 supervisor and the parent shell.
  # Arbitrary root shell access is never granted by this profile: only the
  # add-on's own entrypoint and bounded tool argv vectors exist.
  # ---------------------------------------------------------------------------
  # Scope sends to processes confined by this same profile: the S6 supervisor
  # signalling run.sh/Python on shutdown, and the app signalling its own
  # children.  Receives stay unrestricted so the container runtime can stop
  # PID 1 from the host (the sender is unconfined there).
  #
  # A narrower `signal (send) set=(kill,term,int,hup,cont)` rule dropped onto
  # the same line as an unrestricted receive did NOT authorise this: inside the
  # container `kill -TERM <app pid>` returned EPERM, S6 could not stop the
  # service, and Docker had to SIGKILL the container (exit 137).  The
  # peer-scoped form below is what Home Assistant's example profile uses.
  signal (send) peer=ha_agent_gateway,
  signal (send) set=(kill,term,int,hup,cont),
  signal (receive),
}
